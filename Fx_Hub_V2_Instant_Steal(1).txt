local CREDIT_TEXT = "Open sourced and removed key system by " .. CREDIT_LINK

local function copyLink()
	local ok = false
	pcall(function() if type(setclipboard) == "function" then setclipboard(CREDIT_LINK) ok = true end end)
	pcall(function()
		local s = game:GetService("StarterGui")
		if s and s.SetClipboard then s:SetClipboard(CREDIT_LINK) ok = true end
	end)
	return ok
end

	local gui = Instance.new("ScreenGui")
	gui.Name = "FxHubCredit" gui.DisplayOrder = 999999 gui.IgnoreGuiInset = true
	gui.ResetOnSpawn = false gui.Parent = host

	card.Size = UDim2.fromOffset(460, 240) card.BackgroundColor3 = RGB(8, 18, 36)

		l.Size = UDim2.new(1, -48, 0, h) l.Position = UDim2.fromOffset(24, ypos)

	local sub = label(CREDIT_TEXT, 96, 58, 13, RGB(122, 138, 160))
	sub.TextWrapped = true sub.TextYAlignment = Enum.TextYAlignment.Top

	local btn = Instance.new("TextButton")
	btn.Name = "Copy" btn.Size = UDim2.new(1, -48, 0, 40) btn.Position = UDim2.fromOffset(24, 162)
	btn.BackgroundColor3 = RGB(18, 168, 255) btn.BorderSizePixel = 0 btn.AutoButtonColor = false
	btn.Font = Enum.Font.GothamBold btn.TextSize = 14 btn.Text = "COPY DISCORD LINK" btn.TextColor3 = RGB(4, 11, 24)

	local btnStroke = Instance.new("UIStroke") btnStroke.Color = RGB(191, 244, 255) btnStroke.Transparency = 0.35 btnStroke.Thickness = 1 btnStroke.Parent = btn

	label("or press enter", 208, 14, 11, RGB(70, 82, 100), Enum.TextXAlignment.Center)

	local function paint(target, col)
		tween:Create(target, TweenInfo.new(0.15), {BackgroundColor3 = col}):Play()
	end

	btn.MouseEnter:Connect(function() paint(btn, RGB(59, 234, 255)); tween:Create(btnStroke, TweenInfo.new(0.15), {Transparency = 0}):Play() end)
	btn.MouseLeave:Connect(function() paint(btn, RGB(18, 168, 255)); tween:Create(btnStroke, TweenInfo.new(0.15), {Transparency = 0.35}):Play() end)

	card.MouseLeave:Connect(function() tween:Create(stroke, TweenInfo.new(0.15), {Color = RGB(59, 234, 255)}):Play() end)

	local conn
	local function accept()
		if done then return end
		if not copyLink() then
			btn.Text = "copy failed - press enter to retry"
			paint(btn, RGB(196, 58, 78))
			return
		end
		done = true
		btn.Text = "copied - loading"
		if conn then conn:Disconnect() end
	end

	conn = uis.InputBegan:Connect(function(input, focused)
		if done or focused then return end

		if input.UserInputType == Enum.UserInputType.MouseButton1 then accept() end

	tween:Create(bg, fade, {BackgroundTransparency = 1}):Play()

	tween:Create(stroke, fade, {Transparency = 1}):Play()

		elseif d:IsA("Frame") then
			tween:Create(d, fade, {BackgroundTransparency = 1}):Play()
		end
	end

	task.wait(0.45)
	pcall(function() gui:Destroy() end)
end

local function splashClose()
	if not Splash then return end
	local gui = Splash.gui
	local ok = pcall(function()
		if not gui or not gui.Parent then return end
		local backdrop = gui:FindFirstChildWhichIsA("Frame")
		if backdrop then
			TweenService:Create(backdrop, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
				BackgroundTransparency = 1,
			}):Play()
		end
		task.wait(0.32)
		gui:Destroy()
	end)
	if not ok then pcall(function() gui:Destroy() end) end
	Splash = nil
end

local Library = nil

for _, url in ipairs(LIB_SOURCES) do
	local ok, res = pcall(function()
		return game:HttpGet(url .. "?nocache=" .. tostring(os.time()))
	end)
	if ok and type(res) == "string" and #res > 2000 then
		local chunk, cerr = (loadstring or load)(res)
		if chunk then
			local okLib, lib = pcall(chunk)
			if okLib and type(lib) == "table" then
				Library = lib
				break
			end
		else
			warn("[Fx Hub] Compile error: " .. tostring(cerr))
		end
	end
end

-- Fx Hub floating toggle: independent ScreenGui so the button remains visible while minimized.
-- The main interface is detected from the GUI roots created by the UI library.
do
    local UserInputService = game:GetService("UserInputService")
    local TweenService = game:GetService("TweenService")
    local CoreGui = game:GetService("CoreGui")
    local PlayerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")

    local function getGuiHost()
        local host
        pcall(function() if type(gethui) == "function" then host = gethui() end end)
        if typeof(host) == "Instance" then return host end
        local ok, robloxGui = pcall(function() return CoreGui:FindFirstChild("RobloxGui") end)
        if ok and robloxGui then return robloxGui end
        return PlayerGui or CoreGui
    end

    local host = getGuiHost()
    local mainGui
    local function isOurMainGui(obj)
        if not obj:IsA("ScreenGui") then return false end
        if obj.Name == "FxHubSplash" or obj.Name == "FxHubCredit" or obj.Name == "FxHubFloatingToggle" then return false end
        return #obj:GetDescendants() > 8
    end

    -- Prefer the ScreenGui newly created by the library, without assuming its internal name.
    task.wait(0.12)
    for _, root in ipairs(FxGuiRoots) do
        for _, obj in ipairs(root:GetChildren()) do
            if not FxGuiSnapshot[obj] and isOurMainGui(obj) then mainGui = obj end
        end
    end
    if not mainGui then
        for _, obj in ipairs(host:GetChildren()) do
            if isOurMainGui(obj) then mainGui = obj end
        end
        if not mainGui and PlayerGui then
            for _, obj in ipairs(PlayerGui:GetChildren()) do
                if isOurMainGui(obj) then mainGui = obj end
            end
        end
    end

    local floatingGui = Instance.new("ScreenGui")
    floatingGui.Name = "FxHubFloatingToggle"
    floatingGui.ResetOnSpawn = false
    floatingGui.IgnoreGuiInset = true
    floatingGui.DisplayOrder = 1000000
    floatingGui.Parent = host

    local toggle = Instance.new("TextButton")
    toggle.Name = "FxToggle"
    toggle.AnchorPoint = Vector2.new(1, 0.5)
    toggle.Position = UDim2.new(1, -18, 0.58, 0)
    toggle.Size = UDim2.fromOffset(54, 54)
    toggle.BackgroundColor3 = BRAND.Navy
    toggle.BorderSizePixel = 0
    toggle.AutoButtonColor = false
    toggle.Text = "FX"
    toggle.TextColor3 = BRAND.Text
    toggle.TextSize = 17
    toggle.Font = Enum.Font.GothamBold
    toggle.Active = true
    toggle.Draggable = true
    toggle.Parent = floatingGui

    local round = Instance.new("UICorner")
    round.CornerRadius = UDim.new(1, 0)
    round.Parent = toggle
    local outline = Instance.new("UIStroke")
    outline.Color = BRAND.Cyan
    outline.Thickness = 2
    outline.Transparency = 0.12
    outline.Parent = toggle

    local function setMainVisible(visible)
        if mainGui and mainGui.Parent then
            mainGui.Enabled = visible
        else
            -- Fallback: toggle the library's window if its public method exists.
            pcall(function()
                if Window.Toggle then Window:Toggle() end
            end)
        end
    end

    local open = true
    toggle.Activated:Connect(function()
        open = not open
        toggle.Text = open and "FX" or "+"
        TweenService:Create(toggle, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = open and UDim2.fromOffset(54, 54) or UDim2.fromOffset(48, 48),
            BackgroundColor3 = open and BRAND.Navy or BRAND.Royal,
        }):Play()
        setMainVisible(open)
    end)
    toggle.MouseEnter:Connect(function()
        TweenService:Create(outline, TweenInfo.new(0.15), {Thickness = 3, Transparency = 0}):Play()
    end)
    toggle.MouseLeave:Connect(function()
        TweenService:Create(outline, TweenInfo.new(0.15), {Thickness = 2, Transparency = 0.12}):Play()
    end)
end

local function abbreviate(value)
	value = tonumber(value) or 0
	local units = { "", "K", "M", "B", "T", "Qa", "Qi" }
	local index = 1
	while value >= 1000 and index < #units do
		value /= 1000
		index += 1
	end
	if index == 1 then return tostring(math.floor(value)) end
	return string.format("%.2f%s", value, units[index])
end

local function formatClock(seconds)
	seconds = math.max(math.floor(seconds), 0)
	local hours   = math.floor(seconds / 3600)
	local minutes = math.floor((seconds % 3600) / 60)
	local secs    = seconds % 60
	return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

local function packSet(sel)
	if type(sel) ~= "table" then return "" end
	local names = {}
	for k, v in pairs(sel) do
		if type(k) == "string" then names[#names + 1] = k end
		if type(v) == "string" then names[#names + 1] = v end
	end
	table.sort(names)
	return table.concat(names, ",")
end

local function unpackSet(str)
	local set = {}
	if type(str) == "string" and str ~= "" then
		for name in string.gmatch(str, "[^,]+") do
			set[name] = true
		end
	end
	return set
end

local function asList(packed)
	local list = {}
	if type(packed) == "string" and packed ~= "" then
		for name in string.gmatch(packed, "[^,]+") do
			list[#list + 1] = name
		end
	end
	return list
end

local function selHas(sel, value)
	if type(sel) ~= "table" then return true end
	local count, hit = 0, false
	for k, v in pairs(sel) do
		count += 1
		if v == true and k == value then hit = true end
		if v == value then hit = true end
	end
	if count == 0 then return true end
	return hit
end

end

local function startFly()
	local char = getCharacter()
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hum = getHumanoid()
	if not hrp or not hum then return end

	flyOn = true
	setFlag("FlyOn", true)
	pcall(function() hum.PlatformStand = true end)

	flyBG = Instance.new("BodyGyro")
	flyBG.P = 9e4
	flyBG.maxTorque = Vector3.new(9e9, 9e9, 9e9)
	flyBG.cframe = hrp.CFrame
	flyBG.Parent = hrp

	flyBV = Instance.new("BodyVelocity")
	flyBV.velocity = Vector3.zero
	flyBV.maxForce = Vector3.new(9e9, 9e9, 9e9)
	flyBV.Parent = hrp

	setNoclip(true)

	flyConn = RunService.Heartbeat:Connect(function()
		if not flyOn or not isCurrentGen() then stopFly() return end
		local char2 = getCharacter()
		local hrp2 = char2 and char2:FindFirstChild("HumanoidRootPart")
		local cam = workspace.CurrentCamera
		if not hrp2 or not flyBV or not flyBG or not cam then return end

		local move = Vector3.zero
		if flyKeys.W then move += cam.CFrame.LookVector end
		if flyKeys.S then move -= cam.CFrame.LookVector end
		if flyKeys.A then move -= cam.CFrame.RightVector end
		if flyKeys.D then move += cam.CFrame.RightVector end
		if flyKeys.Space then move += Vector3.new(0, 1, 0) end
		if flyKeys.Ctrl then move -= Vector3.new(0, 1, 0) end
		if move.Magnitude > 0 then move = move.Unit * flySpeed end

		flyBV.velocity = move
		flyBG.cframe = cam.CFrame
	end)
end

local function zoneReachable(zone)
	local required = ZoneJumpRequirements[zone]
	if required == nil then return true end

	jp = jp and tonumber(jp.Value) or 0

local function findEggPrompt(mode)
	local map    = workspace:FindFirstChild("Map")
	local stages = map and map:FindFirstChild("Stages")
	if not stages then return nil, nil end

	local root = getRoot()
	local origin = root and root.Position or Vector3.zero
	local zoneFilter   = unpackSet(getFlag("StealZones"))
	local rarityFilter = unpackSet(getFlag("StealRarities"))

	local bestPrompt, bestScore, bestPosition

	for _, stage in ipairs(stages:GetChildren()) do
		local spawned = stage:FindFirstChild("SpawnedEggs")
		if spawned and selHas(zoneFilter, stage.Name) then
			for _, egg in ipairs(spawned:GetChildren()) do
				local rarity = egg:GetAttribute("Rarity")
				if selHas(rarityFilter, rarity) then
					local prompt = egg:FindFirstChildWhichIsA("ProximityPrompt", true)
					if prompt and prompt.Enabled then
						local worldPosition = holderPosition(prompt.Parent)
						if worldPosition then
							local distance = (worldPosition - origin).Magnitude
							local rarityIndex = 0
							if Rarities and Rarities.GetIndex then
								rarityIndex = Rarities.GetIndex(rarity) or 0
							end
							local score
							if mode == "Nearest Egg" then
								score = -distance
							else

								if mode == "Best Egg" and not zoneReachable(stage.Name) then continue end
								score = rarityIndex * 1000000 - distance * 0.001
							end
							if not bestScore or score > bestScore then
								bestPrompt, bestScore, bestPosition = prompt, score, worldPosition
							end
						end
					end
				end
			end
		end
	end

	return bestPrompt, bestPosition
end

local function doCarryRecovery()
	if not Toggles.AutoSteal.Value and not Toggles.AutoPlace.Value then
		CarryWatchSince = nil
		return
	end

	local carried = (tonumber(LocalPlayer:GetAttribute("CarriedEggCount")) or 0) > 0
	local tool = findTool(isEggTool)

	if carried and not tool then
		local base = getPlacementDetector()
		local root = getRoot()
		if base and root and (root.Position - base.Position).Magnitude > 40 then
			teleportTo(base.Position)
			task.wait(0.4)
		end
		if not CarryWatchSince then
			CarryWatchSince = os.clock()
		elseif os.clock() - CarryWatchSince > 30 and DropEggRequest then
			DropEggRequest:FireServer()
			CarryWatchSince = os.clock()
			task.wait(0.5)
		end
	else
		CarryWatchSince = nil
	end
end

local function doAutoPlace()
	if not Toggles.AutoPlace.Value or not PlaceEggRequest then return end
	if os.clock() < NextPlaceAttempt then return end

	local list = {}
	forEachContainer(function(container)
		for _, child in ipairs(container:GetChildren()) do
			if isEggTool(child) then table.insert(list, child) end
		end
	end)
	if #list == 0 then return end

	PlaceRotation += 1
	local carrying = list[(PlaceRotation % #list) + 1]
	local humanoid = getHumanoid()
	local detector = getPlacementDetector()
	if not humanoid or not detector then return end

	NextPlaceAttempt = os.clock() + 2
	pcall(function() humanoid:EquipTool(carrying) end)
	task.wait(0.2)

	PlaceEggRequest:FireServer(carrying:GetAttribute("EggId"), detector.Position + Vector3.new(0, 2, 0))
	task.wait(0.35)

	if not carrying.Parent then
		local delay = tonumber(Options.StealDelay and Options.StealDelay.Value) or 5
		NextStealAt = os.clock() + math.max(delay, 0)
	end
end

local function doAutoSteal()
	if not Toggles.AutoSteal.Value and not QuickStealRequested then return end
	if os.clock() < NextStealAt then return end
	if countEggTools() >= 3 then return end
	if (tonumber(LocalPlayer:GetAttribute("CarriedEggCount")) or 0) > 0 then return end

	local prompt, position = findEggPrompt(Options.StealMode and Options.StealMode.Value)
	if not prompt or not position then return end

	if LocalPlayer:GetAttribute("IsSquatting") == true and StopSquattingRequest then
		StopSquattingRequest:FireServer()
		local deadline = os.clock() + 1.5
		while LocalPlayer:GetAttribute("IsSquatting") == true and os.clock() < deadline do
			task.wait(0.15)
		end
	end

	teleportTo(position)
	task.wait(0.35)
	if prompt.Enabled then fireproximityprompt(prompt) end
	task.wait(0.4)

	local base = getPlacementDetector()
	if base then
		teleportTo(base.Position)
		task.wait(0.6)
	end
	QuickStealRequested = false
end

				end

		end
	end)
	return bestTool

local function findDuplicateAnimal()
	local counts, lowest = {}, {}
	forEachContainer(function(container)
		for _, child in ipairs(container:GetChildren()) do
			if isPetTool(child) then
				local name = child:GetAttribute("AnimalName") or child.Name
				local cps  = tonumber(child:GetAttribute("CashPerSecond")) or math.huge
				counts[name] = (counts[name] or 0) + 1
				if not lowest[name] or cps < lowest[name] then lowest[name] = cps end
			end
		end
	end)

	local bestName, bestCps
	for name, count in pairs(counts) do
		if count >= 2 and (not bestCps or lowest[name] < bestCps) then
			bestName, bestCps = name, lowest[name]
		end
	end
	return bestName
end

local function doAutoMutation()

	local detector  = mutationDetector()

	if Toggles.AutoClaimMutation.Value and active and ready then
		fireMutation("MutationPrompt", detector)
		return
	end

	if Toggles.AutoReturnMutation.Value and active and canRemove then
		fireMutation("MutationRemovePrompt", detector)
		return
	end

	if not Toggles.AutoAddMutation.Value or active then return end

	local target = LocalPlayer:GetAttribute("MutationCraftAnimalName")
	if type(target) ~= "string" or target == "" then
		if Options.MutationMode and Options.MutationMode.Value == "Chosen Animal" then
			target = Options.MutationAnimal and Options.MutationAnimal.Value
		else
			target = findDuplicateAnimal()
		end
	end
	if not target or target == "None" then return end

	local tool = findMutationPet(target)
	if not tool then return end

	local humanoid = getHumanoid()
	if not humanoid then return end

	pcall(function() humanoid:EquipTool(tool) end)
	task.wait(0.2)
	fireMutation("MutationPrompt", detector)
end

			info = info or {}
			info.Side = "Left"
			return addGroupbox(self, info)
		end

		local okBuild, buildErr = pcall(function()
			SaveManager:BuildConfigSection(Tabs.Settings)
		end)

		Tabs.Settings.AddGroupbox = addGroupbox

		if not okBuild then warn("[Fx Hub] config section failed: " .. tostring(buildErr)) end

		SaveManager:LoadAutoloadConfig()
	end)
end

-- Fx Hub V2 tools: lightweight activity log, diagnostics, and emergency stop.
-- This panel is intentionally independent of the UI library so it remains usable on touch devices.
do
    local UserInputService = game:GetService("UserInputService")
    local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:WaitForChild("PlayerGui")
    local old = playerGui:FindFirstChild("FxHubV2Tools")
    if old then pcall(function() old:Destroy() end) end

    local logLines = {}
    local maxLines = 7
    local panelOpen = false
    local ui = Instance.new("ScreenGui")
    ui.Name = "FxHubV2Tools"
    ui.ResetOnSpawn = false
    ui.IgnoreGuiInset = true
    ui.DisplayOrder = 1000001
    ui.Parent = playerGui

    local function make(className, props, parent)
        local obj = Instance.new(className)
        for k, v in pairs(props) do obj[k] = v end
        obj.Parent = parent
        return obj
    end

    local accent = BRAND.Cyan
    local card = make("Frame", {
        Name = "Panel", AnchorPoint = Vector2.new(0, 1),
        Position = UDim2.new(0, 12, 1, -12), Size = UDim2.fromOffset(300, 42),
        BackgroundColor3 = BRAND.Deep, BackgroundTransparency = 0.04,
        BorderSizePixel = 0, ClipsDescendants = true,
    }, ui)
    make("UICorner", {CornerRadius = UDim.new(0, 10)}, card)
    make("UIStroke", {Color = accent, Thickness = 1.5, Transparency = 0.18}, card)

    local header = make("TextLabel", {
        Name = "Header", Position = UDim2.fromOffset(12, 0), Size = UDim2.new(1, -110, 0, 42),
        BackgroundTransparency = 1, Font = Enum.Font.GothamBold, TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left, Text = "FX HUB  /  V2 TOOLS",
        TextColor3 = BRAND.Text,
    }, card)

    local function button(name, text, pos, size, parent)
        local b = make("TextButton", {
            Name = name, Position = pos, Size = size, BackgroundColor3 = BRAND.Mid,
            BorderSizePixel = 0, AutoButtonColor = false, Font = Enum.Font.GothamBold,
            TextSize = 11, Text = text, TextColor3 = BRAND.Text,
        }, parent)
        make("UICorner", {CornerRadius = UDim.new(0, 7)}, b)
        make("UIStroke", {Color = accent, Thickness = 1, Transparency = 0.45}, b)
        b.MouseEnter:Connect(function()
            TweenService:Create(b, TweenInfo.new(0.12), {BackgroundColor3 = BRAND.Royal}):Play()
        end)
        b.MouseLeave:Connect(function()
            TweenService:Create(b, TweenInfo.new(0.12), {BackgroundColor3 = BRAND.Mid}):Play()
        end)
        return b
    end

    local openBtn = button("OpenLog", "LOG", UDim2.new(1, -96, 0, 7), UDim2.fromOffset(40, 28), card)
    local closeBtn = button("Close", "—", UDim2.new(1, -48, 0, 7), UDim2.fromOffset(36, 28), card)
    local logText = make("TextLabel", {
        Name = "LogText", Position = UDim2.fromOffset(12, 48), Size = UDim2.new(1, -24, 0, 92),
        BackgroundColor3 = BRAND.Navy, BackgroundTransparency = 0.1, BorderSizePixel = 0,
        Font = Enum.Font.Code, TextSize = 11, TextColor3 = BRAND.Text,
        TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top,
        TextWrapped = true, Text = "No activity yet.",
    }, card)
    make("UICorner", {CornerRadius = UDim.new(0, 7)}, logText)
    local diagnostics = make("TextLabel", {
        Name = "Diagnostics", Position = UDim2.fromOffset(12, 146), Size = UDim2.new(1, -24, 0, 54),
        BackgroundTransparency = 1, Font = Enum.Font.Gotham, TextSize = 11,
        TextColor3 = BRAND.Sheen, TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top, TextWrapped = true,
        Text = "Diagnostics: press CHECK to inspect the current session.",
    }, card)
    local stopBtn = button("StopAll", "STOP ALL", UDim2.fromOffset(12, 207), UDim2.new(0.5, -18, 0, 28), card)
    stopBtn.BackgroundColor3 = Color3.fromRGB(100, 16, 36)
    local checkBtn = button("Check", "CHECK", UDim2.new(0.5, 6, 0, 207), UDim2.new(0.5, -18, 0, 28), card)

    local function renderLog()
        local first = math.max(1, #logLines - maxLines + 1)
        local out = {}
        for i = first, #logLines do table.insert(out, logLines[i]) end
        logText.Text = #out > 0 and table.concat(out, "\n") or "No activity yet."
    end

    local function logActivity(message)
        table.insert(logLines, os.date("%H:%M:%S") .. "  " .. tostring(message))
        if #logLines > 40 then table.remove(logLines, 1) end
        renderLog()
    end

    local function setExpanded(value)
        panelOpen = value
        openBtn.Text = value and "HIDE" or "LOG"
        local target = value and UDim2.fromOffset(300, 246) or UDim2.fromOffset(300, 42)
        TweenService:Create(card, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = target}):Play()
    end

    openBtn.Activated:Connect(function() setExpanded(not panelOpen) end)
    closeBtn.Activated:Connect(function() setExpanded(false) end)

    local function runDiagnostics()
        local char = LocalPlayer.Character
        local humanoid = char and char:FindFirstChildOfClass("Humanoid")
        local remotesText = tostring(REMOTE_COUNT or 0) .. "/10 remotes"
        local guiText = playerGui and "PlayerGui OK" or "PlayerGui missing"
        local charText = humanoid and "character OK" or "character not ready"
        local libraryText = type(Library) == "table" and "UI library OK" or "UI library unavailable"
        local saveText = SaveManager and "config manager available" or "config manager unavailable"
        diagnostics.Text = table.concat({remotesText, guiText, charText, libraryText, saveText}, "  |  ")
        logActivity("Diagnostics refreshed")
    end

    checkBtn.Activated:Connect(runDiagnostics)
    stopBtn.Activated:Connect(function()
        local stopped = 0
        for name, toggleObj in pairs(Toggles or {}) do
            local ok = pcall(function()
                if toggleObj and toggleObj.SetValue then toggleObj:SetValue(false) end
            end)
            if ok then stopped += 1 end
        end
        pcall(function() if flyOn then stopFly() end end)
        pcall(function() if noclipRunning then setNoclip(false) end end)
        logActivity("Emergency stop requested; toggles processed: " .. tostring(stopped))
        runDiagnostics()
    end)

    -- Record toggle changes without changing their behavior.
    for name, toggleObj in pairs(Toggles or {}) do
        pcall(function()
            if toggleObj and toggleObj.OnChanged then
                toggleObj.OnChanged(function(value)
                    logActivity(tostring(name) .. " = " .. (value and "ON" or "OFF"))
                end)
            end
        end)
    end

    -- Adapt panel width for narrow touch screens.
    if UserInputService.TouchEnabled then
        card.Size = UDim2.new(1, -24, 0, 42)
        card.Position = UDim2.new(0, 12, 1, -12)
        logText.Size = UDim2.new(1, -24, 0, 92)
    end
    logActivity("Fx Hub V2 tools ready")
    runDiagnostics()
end
